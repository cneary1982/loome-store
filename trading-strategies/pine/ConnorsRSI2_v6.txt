// ConnorsRSI2 — Mean-Reversion Strategy (Cutler's RSI)
// Pine Script v6
// Backtest: +106R combined across SPY/QQQ/GLD on 15m (2yr)
//   GLD +50R | SPY +40R | QQQ +16R
//
// IMPORTANT: Use on 15-minute chart
//
// Logic:
//   Uses Cutler's RSI (SMA-based), NOT Wilder's (ta.rsi uses RMA)
//   Long when Cutler RSI(2) crosses below 10 AND close > SMA(50)
//   TP = 2x ATR(14), SL = 1x ATR(14)
//   Long-only mean-reversion — buys extreme dips in uptrends

//@version=6
strategy("ConnorsRSI2 (Cutler)", overlay=true, default_qty_type=strategy.percent_of_equity,
         default_qty_value=10, initial_capital=10000, commission_value=0.0,
         calc_on_every_tick=false)

// ─── Inputs ────────────────────────────────────────────────────────
rsiLen       = input.int(2,     "RSI Period",         minval=1)
rsiBuy       = input.float(10,  "RSI Buy Threshold",  minval=1,  maxval=50)
trendLen     = input.int(50,    "Trend SMA Length",    minval=5)
useTrend     = input.bool(true, "Use Trend Filter")
atrLen       = input.int(14,    "ATR Period",          minval=1)
slMult       = input.float(1.0, "SL x ATR",           minval=0.1, step=0.1)
tpMult       = input.float(2.0, "TP x ATR",           minval=0.1, step=0.1)

// ─── Time Filter (9:30-13:00 ET entry window) ─────────────────────
useTimeFilter = input.bool(true, "Use 9:30-13:00 ET Entry Window")
entryStart    = input.int(930,   "Entry Start (HHMM ET)", minval=0, maxval=2359)
entryEnd      = input.int(1300,  "Entry End (HHMM ET)",   minval=0, maxval=2359)

nyHour    = hour(time, "America/New_York")
nyMinute  = minute(time, "America/New_York")
nyHHMM    = nyHour * 100 + nyMinute
inWindow  = not useTimeFilter or (nyHHMM >= entryStart and nyHHMM < entryEnd)

// ─── Cutler's RSI (SMA-based, matches Python _rsi) ────────────────
// TradingView's ta.rsi() uses Wilder's RMA smoothing.
// The Python backtest uses SMA of gains/losses (Cutler's RSI).
cutlerRsi(src, len) =>
    change = src - src[1]
    gain   = math.max(change, 0.0)
    loss   = math.max(-change, 0.0)
    avgGain = ta.sma(gain, len)
    avgLoss = ta.sma(loss, len)
    if avgLoss == 0
        100.0
    else
        rs = avgGain / avgLoss
        100.0 - (100.0 / (1.0 + rs))

// ─── Indicators ────────────────────────────────────────────────────
rsiVal   = cutlerRsi(close, rsiLen)
trendMA  = ta.sma(close, trendLen)
atrVal   = ta.atr(atrLen)

// ─── Conditions ────────────────────────────────────────────────────
uptrend       = not useTrend or close > trendMA
crossedUnder  = rsiVal < rsiBuy and rsiVal[1] >= rsiBuy
longCondition = crossedUnder and uptrend and inWindow

// ─── Entry (long-only) ────────────────────────────────────────────
if longCondition and strategy.position_size == 0
    sl = close - slMult * atrVal
    tp = close + tpMult * atrVal
    strategy.entry("Long", strategy.long)
    strategy.exit("Exit", "Long", stop=sl, limit=tp)

// ─── Plots ─────────────────────────────────────────────────────────
plot(trendMA, "Trend SMA", color=color.new(color.orange, 0), linewidth=2)

bgcolor(longCondition ? color.new(color.green, 85) : na)

// RSI panel
hline(rsiBuy, "RSI Buy Level", color=color.red, linestyle=hline.style_dashed)
plot(rsiVal, "Cutler RSI(2)", color=color.new(color.blue, 0), display=display.pane)
